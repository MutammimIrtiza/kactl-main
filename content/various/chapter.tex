	\section{Bitwise Identities}

	\subsection*{Some properties of bitwise operations}
	\begin{itemize}
	  \item $a|b = a\oplus b + a\&b$
	  \item $a\oplus(a\&b) = (a|b)\oplus b$
	  \item $b\oplus(a\&b) = (a|b)\oplus a$
	  \item $(a\&b)\oplus(a|b) = a\oplus b$
	\end{itemize}

	\subsection*{Addition}
	\begin{itemize}
	  \item $a+b = a|b + a\&b$
	  \item $a+b = a\oplus b + 2(a\&b)$
	\end{itemize}

	\subsection*{Subtraction}
	\begin{itemize}
	  \item $a-b = (a\oplus(a\&b))-((a|b)\oplus a)$
	  \item $a-b = ((a|b)\oplus b)-((a|b)\oplus a)$
	  \item $a-b = (a\oplus(a\&b))-(b\oplus(a\&b))$
	  \item $a-b = ((a|b)\oplus b)-(b\oplus(a\&b))$
	\end{itemize}
